# Embedded Checkout demo: server templates

Templates for the `demo-embedded-checkout` skill. Before you copy them, check
every Tapaya call against the live docs with the `tapaya-docs` MCP tools. The
docs win if they differ from this file. The browser files are in
[browser.md](browser.md).

Sources:

- https://docs.tapaya.com/online/integration/embedded_checkout
- https://docs.tapaya.com/online/integration/node_sdk#embedded-checkout
  (`checkout.configuration()`, `checkout.sessions.prepare()`, `verify()`)
- https://docs.tapaya.com/online/getting-started/testing (test cards)

The server uses the built-in `node:http` module, `@tapayadot/checkout` for the
Tapaya calls, and Node's `--env-file` flag, because the SDK does not load
`.env` files itself. `esbuild` is a dev dependency that bundles the browser
loader from `@tapayadot/checkout/client` (see "Why a bundler" in browser.md).

## server.mjs

```js
import { createServer } from 'node:http'
import { randomUUID } from 'node:crypto'
import { readFile } from 'node:fs/promises'
import { PaymentsApiError, Tapaya, TapayaError, TapayaOrderMismatchError } from '@tapayadot/checkout'

const PORT = Number(process.env.PORT ?? 4243)
const BASE_URL = `http://localhost:${PORT}`
const CURRENCY = (process.env.DEMO_CURRENCY ?? 'EUR').toUpperCase() // embedded calls need uppercase
const AMOUNT = Number(process.env.DEMO_AMOUNT ?? 2450) // minor units: 2450 = 24.50

// Reads TAPAYA_SECRET_KEY and TAPAYA_ENVIRONMENT. Throws at startup if the key is missing.
const tapaya = new Tapaya()

// Demo only: orders live in memory and disappear on restart.
const orders = new Map()

const escape = (value) => String(value).replace(/[&<>"']/g, (c) => `&#${c.charCodeAt(0)};`)
const money = (amount, currency) => `${currency} ${(amount / 100).toFixed(2)}`
const page = (title, body) => `<!doctype html>
<html lang="en"><head><meta charset="utf-8"><title>${title}</title>
<meta name="viewport" content="width=device-width, initial-scale=1">
<style>body{font-family:system-ui,sans-serif;max-width:32rem;margin:3rem auto;padding:0 1rem}
pre{background:#f4f4f5;padding:1rem;overflow:auto}button{font-size:1rem;padding:.6rem 1.2rem;margin-top:1rem}
#card-fields{min-height:6rem}</style>
</head><body>${body}</body></html>`

function sendHtml(res, status, html, headers = {}) {
  res.writeHead(status, { 'Content-Type': 'text/html; charset=utf-8', 'Cache-Control': 'no-store', ...headers })
  res.end(html)
}

function sendJson(res, status, body) {
  res.writeHead(status, { 'Content-Type': 'application/json', 'Cache-Control': 'no-store' })
  res.end(JSON.stringify(body))
}

function describeError(error) {
  if (error instanceof TapayaOrderMismatchError) return `Order mismatch for payment ${error.sessionId}: ${error.message}`
  if (error instanceof PaymentsApiError) {
    return `HTTP ${error.status} ${error.code ?? ''}: ${error.message} (requestId: ${error.requestId ?? 'n/a'})`
  }
  if (error instanceof TapayaError) return `${error.name}: ${error.message}`
  return String(error)
}

// Stand-in for your app's getCheckoutOrder(): the order comes from a cookie the server set.
function getCheckoutOrder(req) {
  const match = /(?:^|;\s*)demo_order=([^;]+)/.exec(req.headers.cookie ?? '')
  return match ? orders.get(decodeURIComponent(match[1])) : undefined
}

function newOrder() {
  const id = `demo-${Date.now()}-${randomUUID().slice(0, 8)}`
  const order = { id, amount: AMOUNT, currency: CURRENCY, status: 'pending_payment', paymentId: null }
  orders.set(id, order)
  return order
}

// GET / : the checkout page with the card fields container.
function checkoutPage(req, res) {
  let order = getCheckoutOrder(req)
  if (!order || order.status !== 'pending_payment') order = newOrder()
  sendHtml(res, 200, page('Tapaya Embedded Checkout demo', `
    <h1>Demo checkout</h1>
    <p>Order <code>${escape(order.id)}</code>: <strong>${money(order.amount, order.currency)}</strong></p>
    <form id="checkout">
      <div id="card-fields"></div>
      <button type="submit" disabled>Pay ${money(order.amount, order.currency)}</button>
      <p id="message" role="status"></p>
    </form>
    <p><small>Sandbox only. Test card 4242 4242 4242 4242, any future expiry, CVC 123.</small></p>
    <script type="module" src="/checkout.js"></script>`),
    { 'Set-Cookie': `demo_order=${encodeURIComponent(order.id)}; Path=/; HttpOnly; SameSite=Lax` })
}

// GET /api/config : publishable key and API origin. Both are safe for the browser.
let browserConfig
async function config(res) {
  try {
    browserConfig ??= await tapaya.checkout.configuration()
    const { publishableKey, apiOrigin, environment } = browserConfig
    sendJson(res, 200, { publishableKey, apiOrigin, environment })
  } catch (error) {
    browserConfig = undefined
    console.error(error)
    sendJson(res, 502, { error: describeError(error) })
  }
}

// POST /api/payments : the `prepare` endpoint the card fields call (bare POST, cookies, no body).
async function preparePayment(req, res) {
  const order = getCheckoutOrder(req)
  if (!order) return sendJson(res, 404, { error: 'No checkout order' })
  const details = { merchantOrderId: order.id, amount: order.amount, currency: order.currency }
  const payable = order.status === 'pending_payment'
  try {
    const session = payable
      ? await tapaya.checkout.sessions.prepare(details)
      : await tapaya.checkout.sessions.verify(details)
    if (!session) return sendJson(res, 409, { error: 'Order cannot accept payment' })
    console.log('prepare', order.id, '->', session.id, session.status, session.paymentStatus)
    // Never log the client secret.
    sendJson(res, 200, {
      clientSecret: payable ? session.clientSecret : null,
      status: session.status,
      paymentStatus: session.paymentStatus,
    })
  } catch (error) {
    console.error(describeError(error))
    sendJson(res, 502, { error: 'Could not prepare the payment' })
  }
}

// GET /checkout/confirmation : the browser result is for display only; verify on the server.
async function confirmation(req, res) {
  const order = getCheckoutOrder(req)
  if (!order) return sendHtml(res, 404, page('No order', '<p>No order. <a href="/">Start again</a>.</p>'))
  try {
    const session = await tapaya.checkout.sessions.verify({
      merchantOrderId: order.id, amount: order.amount, currency: order.currency,
    })
    const paid = session?.paymentStatus === 'successful'
    if (paid && order.status !== 'paid') { order.status = 'paid'; order.paymentId = session.id } // record once
    sendHtml(res, 200, page('Payment result', `
      <h1>${paid ? 'Paid' : 'Not paid yet'}</h1>
      <pre>${escape(JSON.stringify({
        orderId: order.id,
        paymentId: session?.id ?? null,
        status: session?.status ?? null,
        paymentStatus: session?.paymentStatus ?? null,
        amount: session ? money(session.amount, session.currency) : null,
      }, null, 2))}</pre>
      ${paid ? '' : '<p>If the payment is still <code>pending</code>, refresh in a moment.</p>'}
      <p><a href="/">New order</a></p>`))
  } catch (error) {
    console.error(error)
    sendHtml(res, 502, page('Verification failed', `<pre>${escape(describeError(error))}</pre>`))
  }
}

async function bundle(res) {
  try {
    const js = await readFile(new URL('./public/checkout.js', import.meta.url))
    res.writeHead(200, { 'Content-Type': 'text/javascript; charset=utf-8', 'Cache-Control': 'no-store' })
    res.end(js)
  } catch {
    sendJson(res, 500, { error: 'Run `npm run build` first' })
  }
}

createServer(async (req, res) => {
  const { pathname } = new URL(req.url, BASE_URL)
  if (req.method === 'GET' && pathname === '/') return checkoutPage(req, res)
  if (req.method === 'GET' && pathname === '/checkout.js') return bundle(res)
  if (req.method === 'GET' && pathname === '/api/config') return config(res)
  if (req.method === 'POST' && pathname === '/api/payments') return preparePayment(req, res)
  if (req.method === 'GET' && pathname === '/checkout/confirmation') return confirmation(req, res)
  sendJson(res, 404, { error: 'Not found' })
}).listen(PORT, '127.0.0.1', () => {
  console.log(`Embedded Checkout demo on ${BASE_URL} (environment: ${process.env.TAPAYA_ENVIRONMENT ?? 'from key'})`)
})
```

## Notes on the template

- Embedded payments do not use return URLs, so no Allowed websites setup is
  needed for cards. Apple Pay needs a registered shop domain and is out of
  scope for this demo.
- `prepare()` picks its own idempotency key from the order, so calling it on
  every page load and every submit never creates a second charge.
- `prepare()` and `verify()` throw `TapayaOrderMismatchError` when the order's
  amount or currency changed after payment started. The demo freezes both.
- Fulfill only on `paymentStatus === 'successful'`. `status: 'completed'` also
  covers `'authorized'`, an uncaptured hold.
- The `SameSite=Lax` cookie means a cross-site page cannot make the browser
  send it with a POST. A real app still needs its usual CSRF protection,
  authentication, a database, webhooks, and a job that settles orders whose
  shoppers leave.
- If you add a Content Security Policy, allow `script-src https://js.tapaya.com`
  and `frame-src https://js.tapaya.com`.
